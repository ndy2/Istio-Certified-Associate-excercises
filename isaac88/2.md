```yaml
apiVersion: networking.istio.io/v1

metadata:
  name: bookinfo

  host: bookinfo
  subsets:
  - name: v1
    labels:
      version: v1

kind: VirtualService
metadata:
  name: bookinfo
spec:
  hosts:
    - bookinfo

```

## ex2

```yaml
apiVersion: networking.istio.io/v1
kind: DestinationRule
metadata:
  name: bookinfo
spec:
  host: bookinfo
  subsets:
  - name: v1
    labels:
      version: v1
  - name: v2
    labels:
      version: v2
---
apiVersion: networking.istio.io/v1
kind: VirtualService
metadata:
  name: ratings-route
spec:
  hosts:
    - bookinfo
  http:
    - match:
        - headers:
            end-user:
              exact: jason
      route:
        - destination:
            host: bookinfo
            subset: v2
    - route:
        - destination:
            host: bookinfo
            subset: v1
```
